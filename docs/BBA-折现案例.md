```java
/**
   * @param amt       每期折现前金额
   * @param n 有效经过月份数
   * @param rateMonth  选取利率年月yyyymm
   * @param middle  是否首月折现一半
   * @return 折现后总金额
   */
   private BigDecimal getPvAmt(BigDecimal amt, int n, String rateMonth,Boolean middle) {
    if (BigDecimal.ZERO.compareTo(amt) == 0) {
      return BigDecimal.ZERO;
    }
    Map<Integer, BigDecimal> monthsRateMap = context.getDisRateCache().getOrDefault(valMonth, Collections.emptyMap());
    if (monthsRateMap.isEmpty()){
      throw new RuntimeException(String.format("评估月%s，对应的月利率曲线不存在", valMonth));
    }
    //折现到当前评估时点
    BigDecimal product = BigDecimal.ONE;
    BigDecimal pvAmt = BigDecimal.ZERO;
       //利率map的key是由有序月份数组成的，例如202401对应的利率map为，第一个月远期利率曲线key:1,value:xxx,第二个月远期利率曲线key:2,value:xxx
    for (int i = 0; i < n; i++) {
        if(middle && i=0){
            product = product.multiply(monthsRateMap.get(i + 1).divide(BigDecimal.valueOf(2), MathContext.DECIMAL128).add(BigDecimal.ONE)).setScale(10, RoundingMode.HALF_UP);
        }else{
            product = product.multiply(monthsRateMap.get(i + 1).add(BigDecimal.ONE)).setScale(10, RoundingMode.HALF_UP);
        }     
      BigDecimal result = amt.divide(product, 10, RoundingMode.HALF_UP);
      pvAmt = pvAmt.add(result);
    }
    return pvAmt;
  }
```
